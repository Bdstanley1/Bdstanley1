async function improveStudioRendering(){let env;try{const {RGBELoader}=await import('./RGBELoader.js');const hdr=await new RGBELoader().loadAsync('./assets/studio_small_08_1k.hdr');hdr.mapping=T.EquirectangularReflectionMapping;const pmrem=new T.PMREMGenerator(renderer);env=pmrem.fromEquirectangular(hdr);scene.environment=env.texture;hdr.dispose();pmrem.dispose();}catch(e){console.warn('Optional studio lighting unavailable',e.message);}if(env)scene.traverse(o=>{if(o.isHemisphereLight)o.intensity=.30;if(o.isDirectionalLight)o.intensity*=.43;});const sm=body.material;body.material=new T.MeshPhysicalMaterial({map:sm.map,color:sm.color,bumpMap:sm.bumpMap,bumpScale:sm.bumpScale,roughness:.67,metalness:0,specularIntensity:.38,clearcoat:.035,clearcoatRoughness:.8,envMapIntensity:.85});sm.dispose();const hm=hair.material;hair.material=new T.MeshPhysicalMaterial({map:hm.map,color:hm.color,alphaTest:.35,side:T.DoubleSide,roughness:.63,metalness:0,anisotropy:.65,anisotropyRotation:Math.PI/2,specularIntensity:.55,envMapIntensity:.85});hm.dispose();Object.assign(clothMat,{map:makeHeatherRibMap(),roughness:.89,bumpScale:.0007,envMapIntensity:.8,sheen:.40,sheenRoughness:.88,sheenColor:new T.Color('#bcbab4'),specularIntensity:.25});clothMat.needsUpdate=true;addIrisDetail();renderer.toneMappingExposure=.95;return {environment:!!env,skin:'physical artistic shader',hair:'anisotropic artistic shader',cloth:'heather diffuse + rib bump'};}
function makeHeatherRibMap(){const c=document.createElement('canvas');c.width=c.height=512;const ctx=c.getContext('2d'),im=ctx.createImageData(512,512);let seed=12091;const rng=()=>{seed=(Math.imul(seed,1664525)+1013904223)>>>0;return seed/4294967296};for(let y=0;y<512;y++)for(let x=0;x<512;x++){const phase=x/64,stitch=Math.cos((x%64-32)*.11+Math.abs(y%32-16)*.22),value=Math.max(0,Math.min(255,236+11*Math.cos(2*Math.PI*phase)+3*stitch+9*(rng()-.5))),k=(y*512+x)*4;im.data[k]=im.data[k+1]=im.data[k+2]=value;im.data[k+3]=255;}ctx.putImageData(im,0,0);const t=new T.CanvasTexture(c);t.colorSpace=T.SRGBColorSpace;t.wrapS=t.wrapT=T.RepeatWrapping;t.anisotropy=Math.min(8,renderer.capabilities.getMaxAnisotropy());return t;}
function addIrisDetail(){const c=document.createElement('canvas');c.width=c.height=256;const ctx=c.getContext('2d'),im=ctx.createImageData(256,256);for(let y=0;y<256;y++)for(let x=0;x<256;x++){const dx=(x-128)/128,dy=(y-128)/128,r=Math.hypot(dx,dy),a=Math.atan2(dy,dx),spokes=Math.sin(a*91+r*24)*Math.sin(a*47-r*15)+.45*Math.sin(a*197),pupil=r<.29,limbal=r>.90,base=pupil?8:limbal?24:59+13*spokes+12*Math.cos(r*32),i=(y*256+x)*4;im.data[i]=base*1.30;im.data[i+1]=base*.89;im.data[i+2]=base*.50;im.data[i+3]=255;}ctx.putImageData(im,0,0);const texture=new T.CanvasTexture(c);texture.colorSpace=T.SRGBColorSpace;figure.traverse(o=>{if(!o.isMesh)return;if(o.name.startsWith('Iris ')){const g=o.geometry,p=g.attributes.position,u=g.attributes.uv,box=new T.Box3().setFromBufferAttribute(p),center=box.getCenter(new T.Vector3()),sz=box.getSize(new T.Vector3());for(let i=0;i<p.count;i++)u.setXY(i,(p.getX(i)-center.x)/sz.x+.5,(p.getY(i)-center.y)/sz.y+.5);u.needsUpdate=true;o.material.map=texture;o.material.color.set('#ffffff');o.material.roughness=.35;o.material.needsUpdate=true;}if(o.name.startsWith('Eye white ')){o.material.color.set('#e1d7c8');o.material.roughness=.30;}});}
