function clipSubset(m,fields,offset=0){const pos=[],norm=[],uv=[],si=[],sw=[],ix=[];function at(id){return {p:m.positions.slice(id*3,id*3+3),n:m.normals.slice(id*3,id*3+3),u:m.uv.slice(id*2,id*2+2),i:m.skinIndex.slice(id*4,id*4+4),w:m.skinWeight.slice(id*4,id*4+4)}}function mix(a,b,t){const agg=new Map();for(let k=0;k<4;k++){agg.set(a.i[k],(agg.get(a.i[k])||0)+(1-t)*a.w[k]);agg.set(b.i[k],(agg.get(b.i[k])||0)+t*b.w[k]);}const top=[...agg].filter(q=>q[1]>0).sort((a,b)=>b[1]-a[1]).slice(0,4);const sum=top.reduce((s,q)=>s+q[1],0)||1;while(top.length<4)top.push([0,0]);return {p:a.p.map((v,k)=>v+(b.p[k]-v)*t),n:a.n.map((v,k)=>v+(b.n[k]-v)*t),u:a.u.map((v,k)=>v+(b.u[k]-v)*t),i:top.map(q=>q[0]),w:top.map(q=>q[1]/sum)}}for(let j=0;j<m.indices.length;j+=3){let polygon=m.indices.slice(j,j+3).map(at);for(let field of fields){if(!polygon.length)break;let next=[],prev=polygon[polygon.length-1],a=field(...prev.p);for(let v of polygon){const b=field(...v.p);if((a>=0)!==(b>=0)){let t=a/(a-b);next.push(mix(prev,v,Math.min(1,Math.max(0,t))));}if(b>=0)next.push(v);prev=v;a=b;}polygon=next;}if(polygon.length<3)continue;for(let k=1;k<polygon.length-1;k++)for(let v of [polygon[0],polygon[k],polygon[k+1]]){const n=new T.Vector3(...v.n).normalize();pos.push(v.p[0]+n.x*offset,v.p[1]+n.y*offset,v.p[2]+n.z*offset);norm.push(n.x,n.y,n.z);uv.push(...v.u);si.push(...v.i);sw.push(...v.w);ix.push(ix.length);}}return geometry({positions:pos,normals:norm,uv,indices:ix,skinIndex:si,skinWeight:sw});}
